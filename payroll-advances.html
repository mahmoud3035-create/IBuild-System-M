'use strict'; const express = require('express'); const db = require('./database/db'); /* * إنشاء جداول السلف والأقساط الشهرية إذا لم تكن موجودة */ async function ensureSchema() { await db.query(` CREATE TABLE IF NOT EXISTS payroll_advances ( id BIGINT UNSIGNED NOT NULL AUTO_INCREMENT, employee_id INT NOT NULL, advance_date DATE NOT NULL, amount DECIMAL(12,2) NOT NULL DEFAULT 0, reason VARCHAR(500) NULL, created_at TIMESTAMP NOT NULL DEFAULT CURRENT_TIMESTAMP, PRIMARY KEY (id), KEY idx_pa_emp (employee_id), KEY idx_pa_date (advance_date) ) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 `); await db.query(` CREATE TABLE IF NOT EXISTS payroll_advance_deductions ( id BIGINT UNSIGNED NOT NULL AUTO_INCREMENT, employee_id INT NOT NULL, payroll_month DATE NOT NULL, amount DECIMAL(12,2) NOT NULL DEFAULT 0, reason VARCHAR(500) NULL, created_at TIMESTAMP NOT NULL DEFAULT CURRENT_TIMESTAMP, updated_at TIMESTAMP NOT NULL DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP, PRIMARY KEY (id), UNIQUE KEY uq_pad_emp_month (employee_id, payroll_month), KEY idx_pad_emp (employee_id) ) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 `); } /* * تحويل الشهر إلى YYYY-MM */ function monthValue(value) { const month = String(value || '').slice(0, 7); if (/^\d{4}-\d{2}$/.test(month)) { return month; } return null; } /* * تحويل الرقم بشكل آمن */ function numberValue(value) { const number = Number(value); return Number.isFinite(number) ? number : 0; } /* * تسجيل Routes الخاصة بالسلف */ function registerPayrollAdvanceRoutes(app) { /* * مهم جداً: * * payroll-advances-bootstrap.js يقوم بتسجيل Routes * السلف أثناء إنشاء Express App. * * لذلك نضع Parser هنا قبل Routes السلف حتى req.body * يكون متاحاً حتى لو express.json() في server.js * تم تسجيله بعد ذلك. */ app.use( '/api/payroll', express.json({ limit: '10mb' }) ); app.use( '/api/payroll', express.urlencoded({ extended: true }) ); // ===================================================== // GET - جميع السلف // ===================================================== app.get('/api/payroll/advances', async (req, res) => { try { await ensureSchema(); const employeeId = numberValue( req.query.employee_id ); let query = ` SELECT id, employee_id, advance_date, amount, reason, created_at FROM payroll_advances `; const params = []; if (employeeId) { query += ` WHERE employee_id = ? `; params.push(employeeId); } query += ` ORDER BY advance_date DESC, id DESC `; const [rows] = await db.query( query, params ); return res.json({ success: true, advances: rows }); } catch (error) { console.error( 'GET PAYROLL ADVANCES ERROR:', error ); return res.status(500).json({ success: false, message: 'تعذر تحميل سلف الموظف', error: error.message }); } }); // ===================================================== // GET - ملخص السلف // ===================================================== app.get('/api/payroll/advance-summary', async (req, res) => { try { await ensureSchema(); const payrollMonth = monthValue( req.query.month ); const params = []; let monthFilter = ''; if (payrollMonth) { monthFilter = ` AND d.payroll_month = ? `; params.push( `${payrollMonth}-01` ); } const [rows] = await db.query(` SELECT e.id AS employee_id, e.employee_code, e.full_name, COALESCE( a.total_advance, 0 ) AS total_advance, COALESCE( p.total_repaid, 0 ) AS total_repaid, GREATEST( 0, COALESCE(a.total_advance, 0) - COALESCE(p.total_repaid, 0) ) AS balance, COALESCE( d.current_deduction, 0 ) AS current_deduction, COALESCE( d.current_reason, '' ) AS current_deduction_reason FROM employees e INNER JOIN ( SELECT employee_id, SUM(amount) AS total_advance FROM payroll_advances GROUP BY employee_id ) a ON a.employee_id = e.id LEFT JOIN ( SELECT employee_id, SUM(amount) AS total_repaid FROM payroll_advance_deductions GROUP BY employee_id ) p ON p.employee_id = e.id LEFT JOIN ( SELECT employee_id, SUM(amount) AS current_deduction, MAX(reason) AS current_reason FROM payroll_advance_deductions WHERE 1 = 1 ${monthFilter} GROUP BY employee_id ) d ON d.employee_id = e.id ORDER BY e.full_name `, params); const summary = rows.map(row => ({ ...row, total_advance: Number( Number(row.total_advance || 0).toFixed(2) ), total_repaid: Number( Number(row.total_repaid || 0).toFixed(2) ), balance: Number( Number(row.balance || 0).toFixed(2) ), current_deduction: Number( Number(row.current_deduction || 0).toFixed(2) ) })); return res.json({ success: true, summary }); } catch (error) { console.error( 'GET PAYROLL ADVANCE SUMMARY ERROR:', error ); return res.status(500).json({ success: false, message: 'تعذر تحميل ملخص السلف', error: error.message }); } }); // ===================================================== // POST - إضافة سلفة // ===================================================== app.post('/api/payroll/advances', async (req, res) => { try { await ensureSchema(); /* * نقرأ البيانات بأكثر من اسم احتياطياً * حتى لا تحصل مشكلة إذا تغير اسم الحقل في الواجهة. */ const body = req.body || {}; const employeeId = numberValue( body.employee_id ?? body.employeeId ?? body.employee ?? body.staff_id ); const amount = numberValue( body.amount ?? body.advance_amount ?? body.advanceAmount ?? body.value ); const reason = String( body.reason ?? body.advance_reason ?? body.advanceReason ?? '' ).trim(); const advanceDate = String( body.advance_date ?? body.advanceDate ?? new Date().toISOString().slice(0, 10) ).slice(0, 10); /* * التحقق من الموظف */ if (!employeeId) { return res.status(400).json({ success: false, message: 'اختر الموظف' }); } /* * التحقق من المبلغ */ if (amount <= 0) { return res.status(400).json({ success: false, message: 'أدخل مبلغ السلفة' }); } /* * التحقق من السبب */ if (!reason) { return res.status(400).json({ success: false, message: 'أدخل سبب السلفة' }); } /* * التأكد أن الموظف موجود */ const [employee] = await db.query( ` SELECT id FROM employees WHERE id = ? LIMIT 1 `, [employeeId] ); if (!employee.length) { return res.status(404).json({ success: false, message: 'الموظف غير موجود' }); } /* * إضافة السلفة */ const [result] = await db.query( ` INSERT INTO payroll_advances ( employee_id, advance_date, amount, reason ) VALUES (?, ?, ?, ?) `, [ employeeId, advanceDate, amount.toFixed(2), reason ] ); return res.status(201).json({ success: true, id: result.insertId, message: 'تمت إضافة السلفة بنجاح' }); } catch (error) { console.error( 'ADD PAYROLL ADVANCE ERROR:', error ); return res.status(500).json({ success: false, message: 'تعذر إضافة السلفة', error: error.message }); } }); // ===================================================== // POST - حفظ القسط / الخصم الشهري // ===================================================== app.post( '/api/payroll/advance-deduction', async (req, res) => { try { await ensureSchema(); const body = req.body || {}; const employeeId = numberValue( body.employee_id ?? body.employeeId ?? body.employee ?? body.staff_id ); const amount = numberValue( body.amount ?? body.deduction_amount ?? body.deductionAmount ); const payrollMonth = monthValue( body.payroll_month ?? body.payrollMonth ?? body.month ); const reason = String( body.reason ?? body.deduction_reason ?? body.deductionReason ?? '' ).trim(); /* * التحقق من البيانات */ if ( !employeeId || !payrollMonth || amount < 0 ) { return res.status(400).json({ success: false, message: 'بيانات خصم السلفة غير صحيحة' }); } /* * إذا كان هناك خصم لازم يكون له سبب */ if ( amount > 0 && !reason ) { return res.status(400).json({ success: false, message: 'سبب خصم السلفة مطلوب' }); } /* * إجمالي السلف */ const [advanceRows] = await db.query( ` SELECT COALESCE(SUM(amount), 0) AS total FROM payroll_advances WHERE employee_id = ? `, [employeeId] ); /* * إجمالي الخصومات السابقة * باستثناء الشهر الحالي */ const [paidRows] = await db.query( ` SELECT COALESCE(SUM(amount), 0) AS total FROM payroll_advance_deductions WHERE employee_id = ? AND payroll_month <> ? `, [ employeeId, `${payrollMonth}-01` ] ); const totalAdvance = numberValue( advanceRows[0]?.total ); const totalPaid = numberValue( paidRows[0]?.total ); const balance = Math.max( 0, totalAdvance - totalPaid ); /* * لا تسمح بخصم أكبر من المتبقي */ if ( amount > balance + 0.005 ) { return res.status(400).json({ success: false, message: `قيمة الخصم أكبر من رصيد السلف المتبقي (${balance.toFixed(2)} AED)` }); } /* * حفظ القسط للشهر * * إذا كان موجوداً لنفس الموظف ونفس الشهر * يتم تحديثه بدلاً من إنشاء سجل جديد. */ await db.query( ` INSERT INTO payroll_advance_deductions ( employee_id, payroll_month, amount, reason ) VALUES (?, ?, ?, ?) ON DUPLICATE KEY UPDATE amount = VALUES(amount), reason = VALUES(reason) `, [ employeeId, `${payrollMonth}-01`, amount.toFixed(2), reason || null ] ); /* * البحث عن سجل راتب الموظف لنفس الشهر */ const [payrollRows] = await db.query( ` SELECT id, payroll_salary, overtime_amount, additions, deductions, absence_deduction FROM payroll_records WHERE employee_id = ? AND DATE_FORMAT( payroll_month, '%Y-%m' ) = ? LIMIT 1 `, [ employeeId, payrollMonth ] ); /* * إذا كان كشف الراتب موجوداً * يتم تطبيق الخصم تلقائياً. */ if (payrollRows.length) { const row = payrollRows[0]; const baseNet = numberValue(row.payroll_salary) + numberValue(row.overtime_amount) + numberValue(row.additions) - numberValue(row.deductions) - numberValue(row.absence_deduction); const netSalary = Math.max( 0, baseNet - amount ); await db.query( ` UPDATE payroll_records SET net_salary = ? WHERE id = ? `, [ netSalary.toFixed(2), row.id ] ); } return res.json({ success: true, message: 'تم حفظ خصم السلفة لهذا الشهر وربطه بالراتب', amount: Number( amount.toFixed(2) ) }); } catch (error) { console.error( 'SAVE PAYROLL ADVANCE DEDUCTION ERROR:', error ); return res.status(500).json({ success: false, message: 'تعذر حفظ خصم السلفة', error: error.message }); } } ); // ===================================================== // DELETE - حذف سلفة // ===================================================== app.delete( '/api/payroll/advances/:id', async (req, res) => { try { await ensureSchema(); const id = numberValue( req.params.id ); if (!id) { return res.status(400).json({ success: false, message: 'رقم السلفة غير صحيح' }); } const [result] = await db.query( ` DELETE FROM payroll_advances WHERE id = ? `, [id] ); if (!result.affectedRows) { return res.status(404).json({ success: false, message: 'السلفة غير موجودة' }); } return res.json({ success: true, message: 'تم حذف السلفة' }); } catch (error) { console.error( 'DELETE PAYROLL ADVANCE ERROR:', error ); return res.status(500).json({ success: false, message: 'تعذر حذف السلفة', error: error.message }); } } ); } module.exports = registerPayrollAdvanceRoutes;
